		if isfile(SAVE_FILE) then
			return HttpService:JSONDecode(readfile(SAVE_FILE))
		end

			if type(data[k]) == type(def) then CFG[k] = data[k] end
		end
	end

	return (pcall(function()
		writefile(SAVE_FILE, HttpService:JSONEncode(CFG))
	end))
end

	saveToken += 1
	local my = saveToken
	task.delay(0.6, function()
		if alive and my == saveToken then writeNow() end
	end)
end

local function make(class, props, parent)
	local o = Instance.new(class)
	-- bọc pcall từng thuộc tính: 1 thuộc tính sai/không hợp lệ chỉ bị bỏ qua,
	-- không làm dừng (crash) toàn bộ script giữa chừng như trước đây.
	for k, v in pairs(props) do
		pcall(function() o[k] = v end)
	end
	o.Parent = parent
	return o
end

-- gọi HTTP GET, trả về nội dung dạng text (hoặc nil nếu lỗi/không có hàm hỗ trợ).
-- Thử qua nhiều tên hàm phổ biến của executor trước, không có thì dùng game:HttpGet.
local function httpGetText(url)
	local candidates = { (syn and syn.request), http_request, request }
	for _, fn in ipairs(candidates) do
		if type(fn) == "function" then
			local ok, res = pcall(fn, { Url = url, Method = "GET" })
			if ok and res and (res.StatusCode == 200 or res.Success) and res.Body then
				return res.Body
			end
		end
	end
	if game and game.HttpGet then
		local ok, res = pcall(function() return game:HttpGet(url) end)
		if ok and res and #res > 0 then return res end
	end
	return nil
end

-- TÍNH NĂNG: ĐỔI SANG SERVER ÍT NGƯỜI (Server Hop)
----------------------------------------------------------------
-- Dùng API công khai của Roblox để liệt kê server rồi chọn server ít người nhất
-- (trừ server đang đứng), sau đó TeleportService đưa mình qua.
local function findLowPopServer()
	local url = ("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100"):format(game.PlaceId)
	local body = httpGetText(url)
	if not body then return nil, "Không gọi được API (kiểm tra mạng hoặc executor có hỗ trợ request không)" end

	local ok, data = pcall(function() return HttpService:JSONDecode(body) end)
	if not ok or not data or type(data.data) ~= "table" then
		return nil, "Không đọc được danh sách server"
	end

	local best = nil
	for _, srv in ipairs(data.data) do
		if srv.id and srv.id ~= game.JobId and type(srv.playing) == "number" and type(srv.maxPlayers) == "number"
			and srv.playing < srv.maxPlayers then
			if not best or srv.playing < best.playing then
				best = srv
			end
		end
	end

	if not best then return nil, "Không tìm thấy server nào phù hợp (có thể do trang đầu toàn server đầy)" end
	return best, nil
end

local function hopToLowPopServer()
	say("Đang tìm server ít người...")
	task.spawn(function()
		local best, err = findLowPopServer()
		if not alive then return end
		if not best then
			setStatus("Server Hop: " .. (err or "lỗi không rõ"))
			return
		end
		setStatus(("Tìm thấy server %d/%d người, đang chuyển..."):format(best.playing, best.maxPlayers))
		task.wait(0.6)
		local ok, errMsg = pcall(function()
			TeleportService:TeleportToPlaceInstance(game.PlaceId, best.id, p)
		end)
		if not ok and alive then
			setStatus("Chuyển server lỗi: " .. tostring(errMsg))
		end
	end)
end

-- kéo thả dùng được cả chuột lẫn cảm ứng; onClick chỉ chạy khi nhấn mà không kéo
local function makeDraggable(obj, handle, onClick)
	local dragging, moved, startMouse, startPos = false, false, nil, nil

	track(handle.InputBegan:Connect(function(input)
		local t = input.UserInputType
		if t ~= Enum.UserInputType.MouseButton1 and t ~= Enum.UserInputType.Touch then return end
		dragging, moved = true, false
		startMouse, startPos = input.Position, obj.Position

		local c
		c = input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				c:Disconnect()
				dragging = false
				if not moved and onClick then onClick() end
			end
		end)
	end))

	track(UIS.InputChanged:Connect(function(input)
		if not dragging then return end
		local t = input.UserInputType
		if t ~= Enum.UserInputType.MouseMovement and t ~= Enum.UserInputType.Touch then return end
		local d = input.Position - startMouse
		if not moved and d.Magnitude > 6 then moved = true end
		if moved then
			obj.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + d.X,
				startPos.Y.Scale, startPos.Y.Offset + d.Y
			)
		end
	end))
end

-- giải mã base64 -> chuỗi byte (dùng để ghi ảnh PNG nhúng sẵn ra file)
local B64_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
local function b64decode(s)
	local map = {}
	for i = 1, #B64_CHARS do map[B64_CHARS:byte(i)] = i - 1 end
	local out, n = {}, 0
	local buf, bits = 0, 0
	for i = 1, #s do
		local v = map[s:byte(i)]
		if v then
			buf = buf * 64 + v
			bits += 6
			if bits >= 8 then
				bits -= 8
				local div = 2 ^ bits
				local byte = math.floor(buf / div)
				buf = buf - byte * div
				n += 1
				out[n] = string.char(byte)
			end
		end
	end
	return table.concat(out)
end

-- ghi 1 ảnh PNG (dạng base64) ra file tạm rồi trả về asset id để gán vào Image.
-- Dùng chung cho cả logo lẫn banner.
local function loadEmbeddedImage(fileName, b64Data)
	local getAsset = getcustomasset or getsynasset
	if type(writefile) ~= "function" or type(getAsset) ~= "function" then return "" end
	local ok, res = pcall(function()
		writefile(fileName, b64decode(b64Data))
		return getAsset(fileName)
	end)
	if ok and type(res) == "string" and res ~= "" then return res end
	return ""
end

-- phân loại: nhân vật người chơi khác = "Player", model có Humanoid còn lại = "NPC"
local function getKind(model)
	if not model:IsA("Model") or model == p.Character then return nil end
	if not model:FindFirstChildOfClass("Humanoid") then return nil end
	if Players:GetPlayerFromCharacter(model) then
		if CFG.espPlayer then return "Player" end
	else
		if CFG.espNPC then return "NPC" end
	end
	return nil
end

local function mainPart(model)
	return model:FindFirstChild("HumanoidRootPart")
		or model.PrimaryPart
		or model:FindFirstChild("Head")
		or model:FindFirstChildWhichIsA("BasePart", true)
end

local function isTeammate(data)
	if not CFG.teamCheck then return false end
	local plr = data.Plr
	return plr ~= nil and p.Team ~= nil and plr.Team == p.Team
end

local function removeTarget(model)
	local data = ESP.Tracked[model]
	if not data then return end
	data.Highlight:Destroy()
	data.Gui:Destroy()
	ESP.Tracked[model] = nil
end

local function addTarget(model)
	if ESP.Tracked[model] then return end
	local kind = getKind(model)
	if not kind then return end
	local part = mainPart(model)
	if not part then return end

	local color = kind == "Player" and ESP_CONFIG.PlayerColor or ESP_CONFIG.NPCColor

	local h = make("Highlight", {
		Adornee = model,
		DepthMode = Enum.HighlightDepthMode.AlwaysOnTop,
		FillColor = color,
		OutlineColor = Color3.new(1, 1, 1),
		FillTransparency = ESP_CONFIG.FillTransparency,
	}, model)

	local bb = make("BillboardGui", {
		Adornee = model:FindFirstChild("Head") or part,
		AlwaysOnTop = true,
		Size = UDim2.fromOffset(140, 30),
		StudsOffset = Vector3.new(0, 2.5, 0),
	}, espFolder)

	local lbl = make("TextLabel", {
		BackgroundTransparency = 1,
		Size = UDim2.fromScale(1, 1),
		Font = Enum.Font.GothamBold,
		TextSize = 12,
		TextColor3 = Color3.new(1, 1, 1),
		TextStrokeTransparency = 0.3,
	}, bb)

	ESP.Tracked[model] = {
		Highlight = h, Gui = bb, Label = lbl,
		Hum = model:FindFirstChildOfClass("Humanoid"),
		Plr = Players:GetPlayerFromCharacter(model),
		Root = part, Kind = kind, Color = color, Name = model.Name,
		Shown = true, Dist = 0,
	}
end

local function clearAll()
	for model in pairs(ESP.Tracked) do removeTarget(model) end
	for _, c in ipairs(ESP.Conns) do c:Disconnect() end
	table.clear(ESP.Conns)
end

	local list = {}
	table.insert(ESP.Conns, RunService.Heartbeat:Connect(function(dt)
		acc += dt
		if acc < ESP_CONFIG.UpdateRate then return end
		acc = 0

		local cam = workspace.CurrentCamera
		local myRoot = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
		local origin = myRoot and myRoot.Position or (cam and cam.CFrame.Position) or Vector3.zero
		local maxDist, maxTargets = CFG.espDist, CFG.espMax
		table.clear(list)

		for model, data in pairs(ESP.Tracked) do
			local hum = data.Hum
			if not model.Parent or not hum or not hum.Parent then
				removeTarget(model)
			else
				local root = data.Root
				if not root or not root.Parent then
					root = mainPart(model)
					data.Root = root
				end
				local dist = root and (root.Position - origin).Magnitude or 0
				if hum.Health > 0 and dist <= maxDist and not isTeammate(data) then
					data.Dist = dist
					list[#list + 1] = data
				else
					setShown(data, false)
				end
			end
		end

		-- chỉ vẽ N mục tiêu gần nhất
		if #list > maxTargets then
			table.sort(list, function(a, b) return a.Dist < b.Dist end)
		end

		for i, data in ipairs(list) do
			if i > maxTargets then
				setShown(data, false)
			else
				setShown(data, true)
				local hum = data.Hum
				local hp = math.floor(hum.Health)
				local d = math.floor(data.Dist / 5 + 0.5) * 5  -- làm tròn 5m cho đỡ đổi chữ liên tục
				if data.HP ~= hp or data.D ~= d then
					data.HP, data.D = hp, d
					data.Label.Text = string.format("%s | %d HP | %dm", data.Name, hp, d)
					local ratio = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
					data.Highlight.FillColor = ESP_CONFIG.LowHealthColor:Lerp(data.Color, ratio)
				end
			end
		end
	end))
end

	if CFG.espNPC or CFG.espPlayer then startESP() end

local ITEM = { Tracked = {}, Conns = {}, Gen = 0 }

local ITEM_CFG = {
	ToolColor = Color3.fromRGB(255, 215, 0),
	PromptColor = Color3.fromRGB(255, 150, 50),
	UpdateRate = 0.25,
}

local function itemPart(inst, kind)
	if kind == "Tool" then
		local h = inst:FindFirstChild("Handle")
		if h and h:IsA("BasePart") then return h end
		return inst:FindFirstChildWhichIsA("BasePart", true)
	end
	local par = inst.Parent
	if not par then return nil end
	if par:IsA("BasePart") then return par end
	if par:IsA("Attachment") and par.Parent and par.Parent:IsA("BasePart") then
		return par.Parent
	end
	if par:IsA("Model") then
		return par.PrimaryPart or par:FindFirstChildWhichIsA("BasePart", true)
	end
	return nil
end

local function itemName(inst, kind)
	if kind == "Prompt" then
		local t = inst.ObjectText
		if t == "" then t = inst.Parent and inst.Parent.Name or "Item" end
		return t
	end
	return inst.Name
end

-- Tìm "giá trị" của vật phẩm: game nào cũng lưu khác nhau nên thử nhiều chỗ
-- (Attribute, NumberValue/IntValue/StringValue con, ActionText của prompt)
local VALUE_KEYS = {
	"value", "price", "cost", "worth", "sell", "amount",
	"rarity", "tier", "level", "coin", "money", "cash", "gold", "gem",
}

local function fmtVal(v)
	if type(v) == "number" then
		if v >= 1e9 then return string.format("%.1fB", v / 1e9) end
		if v >= 1e6 then return string.format("%.1fM", v / 1e6) end
		if v >= 1e4 then return string.format("%.1fK", v / 1e3) end
		return tostring(math.floor(v * 100 + 0.5) / 100)
	end
	return tostring(v)
end

local function matchKey(name)
	local ln = name:lower()
	for _, k in ipairs(VALUE_KEYS) do
		if ln:find(k, 1, true) then return true end
	end
	return false
end

local function findValue(root)
	-- 1) Attribute có tên giống giá trị
	for name, v in pairs(root:GetAttributes()) do
		if (type(v) == "number" or type(v) == "string") and matchKey(name) then
			return name, v
		end
	end
	-- 2) Value con có tên giống giá trị
	local fallbackName, fallbackVal
	for _, c in ipairs(root:GetChildren()) do
		if c:IsA("NumberValue") or c:IsA("IntValue") or c:IsA("StringValue") then
			if matchKey(c.Name) then return c.Name, c.Value end
			if not fallbackName and not c:IsA("StringValue") then
				fallbackName, fallbackVal = c.Name, c.Value
			end
		end
	end
	-- 3) không khớp tên thì lấy Value số đầu tiên
	return fallbackName, fallbackVal
end

local function itemValueText(data)
	local inst = data.Inst
	local roots = { inst }
	if data.Kind == "Tool" then
		local h = inst:FindFirstChild("Handle")
		if h then roots[#roots + 1] = h end
	else
		local par = inst.Parent
		if par then roots[#roots + 1] = par end
		local m = inst:FindFirstAncestorWhichIsA("Model")
		if m and m ~= par then roots[#roots + 1] = m end
	end

	for _, r in ipairs(roots) do
		local name, v = findValue(r)
		if name ~= nil then return name .. ": " .. fmtVal(v) end
	end

	-- nhiều game ghi giá trong ActionText của prompt (vd "Buy $100")
	if data.Kind == "Prompt" then
		local a = inst.ActionText
		if a ~= "" and a:lower() ~= "interact" then return a end
	elseif inst.ToolTip ~= "" then
		return inst.ToolTip
	end
	return nil
end

-- Tool đang được cầm (nằm trong nhân vật) thì không tính là đồ rơi
local function itemValid(data)
	local inst = data.Inst
	if data.Kind == "Tool" then
		local par = inst.Parent
		return par ~= nil and not par:FindFirstChildOfClass("Humanoid")
	end
	return inst.Enabled
end

local function removeItem(inst)
	local data = ITEM.Tracked[inst]
	if not data then return end
	data.Gui:Destroy()
	ITEM.Tracked[inst] = nil
end

local function addItem(inst)
	if ITEM.Tracked[inst] then return end
	local kind
	if inst.ClassName == "Tool" then
		if not CFG.espItem then return end
		kind = "Tool"
	elseif inst.ClassName == "ProximityPrompt" then
		if not CFG.espPrompt then return end
		if inst:FindFirstAncestorOfClass("Tool") then return end
		local m = inst:FindFirstAncestorWhichIsA("Model")
		if m and m:FindFirstChildOfClass("Humanoid") then return end
		kind = "Prompt"
	else
		return
	end

	local part = itemPart(inst, kind)
	local bb = make("BillboardGui", {
		Adornee = part,
		AlwaysOnTop = true,
		Enabled = false,
		Size = UDim2.fromOffset(220, 20),
		StudsOffset = Vector3.new(0, 1.5, 0),
	}, espFolder)
	local lbl = make("TextLabel", {
		BackgroundTransparency = 1,
		Size = UDim2.fromScale(1, 1),
		Font = Enum.Font.GothamBold,
		TextSize = 11,
		TextColor3 = kind == "Tool" and ITEM_CFG.ToolColor or ITEM_CFG.PromptColor,
		TextStrokeTransparency = 0.3,
	}, bb)

	ITEM.Tracked[inst] = {
		Inst = inst, Kind = kind, Part = part,
		Gui = bb, Label = lbl, Shown = false, Dist = 0,
	}
end

local function setItemShown(data, on)
	if data.Shown ~= on then
		data.Shown = on
		data.Gui.Enabled = on
	end
end

local function clearItems()
	for inst in pairs(ITEM.Tracked) do removeItem(inst) end
	for _, c in ipairs(ITEM.Conns) do c:Disconnect() end
	table.clear(ITEM.Conns)
end

local function startItems()
	local gen = ITEM.Gen

	-- quét nền, chia nhỏ từng đợt
	task.spawn(function()
		local n = 0
		for _, v in ipairs(workspace:GetDescendants()) do
			if ITEM.Gen ~= gen then return end
			local c = v.ClassName
			if c == "Tool" or c == "ProximityPrompt" then addItem(v) end
			n += 1
			if n % 2000 == 0 then task.wait() end
		end
	end)

	-- vật phẩm mới xuất hiện / đồ bị vứt xuống
	table.insert(ITEM.Conns, workspace.DescendantAdded:Connect(function(v)
		local c = v.ClassName
		if c ~= "Tool" and c ~= "ProximityPrompt" then return end
		task.defer(function()
			if ITEM.Gen == gen then addItem(v) end
		end)
	end))

	-- vật phẩm bị nhặt / xoá
	table.insert(ITEM.Conns, workspace.DescendantRemoving:Connect(function(v)
		if ITEM.Tracked[v] then removeItem(v) end
	end))

	local acc, list = 0, {}
	table.insert(ITEM.Conns, RunService.Heartbeat:Connect(function(dt)
		acc += dt
		if acc < ITEM_CFG.UpdateRate then return end
		acc = 0

		local cam = workspace.CurrentCamera
		local myRoot = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
		local origin = myRoot and myRoot.Position or (cam and cam.CFrame.Position) or Vector3.zero
		table.clear(list)

		for inst, data in pairs(ITEM.Tracked) do
			if not inst.Parent then
				removeItem(inst)
			else
				local part = data.Part
				if not part or not part.Parent then
					part = itemPart(inst, data.Kind)
					data.Part = part
					data.Gui.Adornee = part
				end
				if part and itemValid(data) then
					local dist = (part.Position - origin).Magnitude
					if dist <= CFG.espDist then
						data.Dist = dist
						list[#list + 1] = data
					else
						setItemShown(data, false)
					end
				else
					setItemShown(data, false)
				end
			end
		end

		table.sort(list, function(a, b) return a.Dist < b.Dist end)

		for i, data in ipairs(list) do
			if i > CFG.itemMax then
				setItemShown(data, false)
			else
				setItemShown(data, true)
				local d = math.floor(data.Dist / 5 + 0.5) * 5
				local nm = itemName(data.Inst, data.Kind)
				local val = itemValueText(data)
				local txt
				if val then
					txt = string.format("%s [%s] | %dm", nm, val, d)
				else
					txt = string.format("%s | %dm", nm, d)
				end
				if data.Label.Text ~= txt then data.Label.Text = txt end
			end
		end
	end))
end

function ITEM.Refresh()
	ITEM.Gen += 1
	clearItems()
	if CFG.espItem or CFG.espPrompt then startItems() end
end

function ITEM.CountKind(kind)
	local n = 0
	for _, d in pairs(ITEM.Tracked) do
		if d.Kind == kind then n += 1 end
	end
	return n
end

-- TÍNH NĂNG: SPEED / JUMP (ghi đè thuộc tính Humanoid)
----------------------------------------------------------------
local function getHum()
	local c = p.Character
	return c and c:FindFirstChildOfClass("Humanoid")
end

-- resolve(hum, v) trả về (tên thuộc tính, giá trị cần đặt)
-- watch: các thuộc tính cần theo dõi để đặt lại khi game đổi
local function newHumFeature(onKey, valKey, watch, resolve)
	local F = { Conns = {}, Hum = nil, Prop = nil, Orig = nil }

	local function enforce(hum)
		local prop, val = resolve(hum, CFG[valKey])
		if math.abs(hum[prop] - val) > 0.01 then hum[prop] = val end
	end

	-- nhớ giá trị gốc (chỉ 1 lần cho mỗi Humanoid) để trả lại khi tắt
	local function capture(hum)
		if F.Hum == hum then return end
		F.Hum = hum
		F.Prop = (resolve(hum, CFG[valKey]))
		F.Orig = hum[F.Prop]
	end

	local function disconnectAll()
		for _, c in ipairs(F.Conns) do c:Disconnect() end
		table.clear(F.Conns)
	end

	-- gắn vào Humanoid mới (lúc hồi sinh)
	function F.Bind(hum)
		disconnectAll()
		if not hum then return end
		if CFG[onKey] then
			capture(hum)
			enforce(hum)
		end

		local n, t0 = 0, os.clock()
		local function onChange()
			if not CFG[onKey] then return end
			local now = os.clock()
			if now - t0 > 1 then n, t0 = 0, now end
			n += 1
			if n > 120 then return end  -- game ghi đè liên tục: bỏ cuộc để khỏi lag
			enforce(hum)
		end
		for _, name in ipairs(watch) do
			table.insert(F.Conns, hum:GetPropertyChangedSignal(name):Connect(onChange))
		end
	end

	function F.Set(on)
		local hum = getHum()
		if not hum then return end
		if on then
			capture(hum)
			enforce(hum)
		elseif F.Hum == hum and F.Prop and F.Orig then
			hum[F.Prop] = F.Orig
		end
	end

	-- gọi khi đổi giá trị lúc đang bật
	function F.Update()
		local hum = getHum()
		if hum and CFG[onKey] then
			capture(hum)
			enforce(hum)
		end
	end

	-- kiểm tra xem game có đang ghi đè không
	function F.Verify(label)
		task.delay(1.5, function()
			local hum = getHum()
			if not alive or not hum or not CFG[onKey] then return end
			local prop, val = resolve(hum, CFG[valKey])
			if math.abs(hum[prop] - val) > 0.5 then
				setStatus(label .. ": game đang ghi đè, có thể không hiệu quả")
			end
		end)
	end

	function F.Stop()
		F.Set(false)
		disconnectAll()
	end

	return F
end

local SpeedF = newHumFeature("speedOn", "speedVal", { "WalkSpeed" }, function(hum, v)
	return "WalkSpeed", v
end)

local JumpF = newHumFeature("jumpOn", "jumpVal", { "JumpPower", "JumpHeight", "UseJumpPower" }, function(hum, v)
	if hum.UseJumpPower then return "JumpPower", v end
	return "JumpHeight", (v * v) / (2 * workspace.Gravity)  -- đổi JumpPower sang JumpHeight
end)

-- Noclip: tắt va chạm các bộ phận của nhân vật, nhớ để trả lại khi tắt
local NC = { Conn = nil, Saved = {} }

function NC.Set(on)
	if NC.Conn then NC.Conn:Disconnect() NC.Conn = nil end
	if on then
		NC.Conn = RunService.Stepped:Connect(function()
			local c = p.Character
			if not c then return end
			for _, v in ipairs(c:GetChildren()) do
				if v:IsA("BasePart") and v.CanCollide then
					NC.Saved[v] = true
					v.CanCollide = false
				end
			end
		end)
	else
		for part in pairs(NC.Saved) do
			if part.Parent then part.CanCollide = true end
		end
		table.clear(NC.Saved)
	end
end

-- Đi bộ an toàn tới 1 toạ độ: tạm bật Noclip trong lúc di chuyển (nếu chưa bật sẵn) để
-- không bị kẹt địa hình, chuyển Humanoid sang trạng thái Physics để tránh giật (xem lý do
-- trong phần TP khu vực trước đây), xong tự trả mọi thứ về trạng thái cũ khi tới nơi.
local function safeWalkToPos(targetPos, label)
	task.spawn(function()
		local char = p.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if not hrp or not hum then
			say("TP: chưa có nhân vật")
			return
		end

		local wasNoclip = CFG.noclip
		if not wasNoclip then NC.Set(true) end
		say("Đang TP tới " .. label .. "...")

		hum:ChangeState(Enum.HumanoidStateType.Physics)

		local WALK_SPEED = 130
		local conn
		conn = RunService.Heartbeat:Connect(function(dt)
			if not alive or not hrp.Parent then
				if conn then conn:Disconnect() conn = nil end
				return
			end
			local dir = targetPos - hrp.Position
			local dist = dir.Magnitude
			if dist <= 3 then
				if conn then conn:Disconnect() conn = nil end
				return
			end
			local step = math.min(WALK_SPEED * dt, dist)
			hrp.CFrame = CFrame.new(hrp.Position + dir.Unit * step, hrp.Position + dir)
		end)

		while conn do task.wait(0.05) end

		if hrp.Parent then
			hrp.CFrame = CFrame.new(targetPos)
			hum:ChangeState(Enum.HumanoidStateType.GettingUp)
		end

		if not wasNoclip then NC.Set(false) end
		say("Đã tới " .. label)
	end)
end

-- Fly: bay theo hướng camera. PC: Space lên, Ctrl xuống
local FLY = { Conn = nil, Objs = {} }

local function flyStop(restore)
	local wasFlying = #FLY.Objs > 0
	if FLY.Conn then FLY.Conn:Disconnect() FLY.Conn = nil end
	for _, o in ipairs(FLY.Objs) do pcall(function() o:Destroy() end) end
	table.clear(FLY.Objs)
	if restore and wasFlying then
		local hum = getHum()
		if hum then
			hum.PlatformStand = false
			hum:ChangeState(Enum.HumanoidStateType.GettingUp)
		end
	end
end

local function flyStart()
	flyStop(false)
	local char = p.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	local hum = getHum()
	if not hrp or not hum then return false end

	local att = make("Attachment", { Name = "VuongFlyAtt" }, hrp)
	local lv = make("LinearVelocity", {
		Attachment0 = att,
		VelocityConstraintMode = Enum.VelocityConstraintMode.Vector,
		MaxForce = math.huge,
		VectorVelocity = Vector3.zero,
		RelativeTo = Enum.ActuatorRelativeTo.World,
	}, hrp)
	local ao = make("AlignOrientation", {
		Mode = Enum.OrientationAlignmentMode.OneAttachment,
		Attachment0 = att,
		RigidityEnabled = true,
		CFrame = hrp.CFrame,
	}, hrp)
	FLY.Objs = { att, lv, ao }
	hum.PlatformStand = true

	local lastFlat = Vector3.new(0, 0, -1)
	FLY.Conn = RunService.RenderStepped:Connect(function()
		local cam = workspace.CurrentCamera
		if not cam or not hrp.Parent then return end

		local look = cam.CFrame.LookVector
		local flat = Vector3.new(look.X, 0, look.Z)
		if flat.Magnitude > 0.05 then lastFlat = flat.Unit end
		local right = lastFlat:Cross(Vector3.yAxis)

		-- hướng đi (joystick / WASD) chiếu lên hướng nhìn camera
		local md = hum.MoveDirection
		local vel = look * md:Dot(lastFlat) + right * md:Dot(right)
		if UIS:IsKeyDown(Enum.KeyCode.Space) then vel += Vector3.yAxis end
		if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then vel -= Vector3.yAxis end
		if vel.Magnitude > 1 then vel = vel.Unit end

		lv.VectorVelocity = vel * CFG.flySpeed
		ao.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + lastFlat)
	end)
	return true
end

-- TÍNH NĂNG: AUTO NÉ QUÁI (giữ khoảng cách an toàn, quái khó áp sát để đánh trúng)
----------------------------------------------------------------
-- Khác với cách "hồi máu khi bị đánh" (đã bỏ vì không ăn ở game có server kiểm tra máu),
-- cách này chỉ đẩy nhân vật ra xa quái/NPC bằng CFrame — cùng cơ chế với Speed/Fly/Noclip
-- đang dùng tốt trong hub, nên khả năng hoạt động cao hơn nhiều.
local DODGE = { Tracked = {}, Conns = {}, Gen = 0 }
local DODGE_PUSH_SPEED = 50 -- tốc độ né ra xa (stud/giây)

local function dodgeIsMonster(model)
	if not model:IsA("Model") or model == p.Character then return false end
	if Players:GetPlayerFromCharacter(model) then return false end  -- chỉ né quái/NPC, không né người chơi khác
	return model:FindFirstChildOfClass("Humanoid") ~= nil
end

local function dodgeAdd(model)
	if DODGE.Tracked[model] or not dodgeIsMonster(model) then return end
	local root = model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart
	if root then DODGE.Tracked[model] = root end
end

local function dodgeClear()
	table.clear(DODGE.Tracked)
	for _, c in ipairs(DODGE.Conns) do c:Disconnect() end
	table.clear(DODGE.Conns)
end

local function dodgeStart()
	local gen = DODGE.Gen
	task.spawn(function()
		local n = 0
		for _, v in ipairs(workspace:GetDescendants()) do
			if DODGE.Gen ~= gen then return end
			if v.ClassName == "Humanoid" and v.Parent then dodgeAdd(v.Parent) end
			n += 1
			if n % 2000 == 0 then task.wait() end
		end
	end)
	table.insert(DODGE.Conns, workspace.DescendantAdded:Connect(function(v)
		if v.ClassName ~= "Humanoid" then return end
		task.spawn(function()
			local m = v.Parent
			if not m then return end
			m:WaitForChild("HumanoidRootPart", 3)
			if DODGE.Gen == gen then dodgeAdd(m) end
		end)
	end))
	table.insert(DODGE.Conns, workspace.DescendantRemoving:Connect(function(v)
		if DODGE.Tracked[v] then DODGE.Tracked[v] = nil end
	end))
end

function DODGE.Refresh()
	DODGE.Gen += 1
	dodgeClear()
	if CFG.autoDodge then dodgeStart() end
end

-- mỗi khung hình: cộng dồn hướng né từ mọi quái trong bán kính, đẩy ra xa theo hướng đó
track(RunService.Heartbeat:Connect(function(dt)
	if not CFG.autoDodge then return end
	local char = p.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end

	local myPos = hrp.Position
	local radius = CFG.dodgeRadius
	local push = Vector3.zero
	for model, root in pairs(DODGE.Tracked) do
		if root.Parent then
			local diff = myPos - root.Position
			diff = Vector3.new(diff.X, 0, diff.Z)  -- chỉ né theo phương ngang
			local dist = diff.Magnitude
			if dist < radius and dist > 0.1 then
				local strength = (radius - dist) / radius  -- càng gần đẩy càng mạnh
				push = push + diff.Unit * strength
			end
		else
			DODGE.Tracked[model] = nil
		end
	end

	if push.Magnitude > 0.01 then
		hrp.CFrame = hrp.CFrame + push.Unit * (DODGE_PUSH_SPEED * dt)
	end
end))

-- TÍNH NĂNG: FIX LAG (giảm đồ họa để tăng FPS, hoàn tác được)
----------------------------------------------------------------
local LAG = {
	Orig = setmetatable({}, { __mode = "k" }),  -- giá trị gốc của từng vật thể
	OrigQuality = nil,
	OrigTech = nil,
	Conns = {},
	Gen = 0,
}

-- đổi thuộc tính nhưng nhớ giá trị gốc (chỉ nhớ lần đầu) để khôi phục
local function lagSet(inst, prop, value)
	local rec = LAG.Orig[inst]
	if not rec then
		rec = {}
		LAG.Orig[inst] = rec
	end
	if rec[prop] == nil then rec[prop] = inst[prop] end
	inst[prop] = value
end

local function lagProcess(v, level)
	-- mức 1: tắt hiệu ứng nặng (hạt, vệt, tia, lửa, khói, bloom/blur...)
	if v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Beam")
		or v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") or v:IsA("PostEffect") then
		lagSet(v, "Enabled", false)
	elseif level >= 2 then
		-- mức 2: đơn giản hoá vật liệu, bóng, hình dán (bỏ qua nhân vật cho khỏi xấu)
		if v:IsA("BasePart") then
			if not v:IsA("Terrain") and not (v.Parent and v.Parent:FindFirstChildOfClass("Humanoid")) then
				lagSet(v, "Material", Enum.Material.SmoothPlastic)
				lagSet(v, "CastShadow", false)
				lagSet(v, "Reflectance", 0)
			end
		elseif v:IsA("Decal") then
			local model = v.Parent and v.Parent.Parent
			if not (model and model:FindFirstChildOfClass("Humanoid")) then
				lagSet(v, "Transparency", 1)
			end
		end
	end
end

function LAG.Restore()
	LAG.Gen += 1  -- huỷ đợt quét đang chạy nền
	for _, c in ipairs(LAG.Conns) do c:Disconnect() end
	table.clear(LAG.Conns)

	for inst, rec in pairs(LAG.Orig) do
		if inst.Parent then
			for prop, val in pairs(rec) do
				pcall(function() inst[prop] = val end)
			end
		end
	end
	table.clear(LAG.Orig)

	if LAG.OrigQuality ~= nil then
		pcall(function()
			UserSettings():GetService("UserGameSettings").SavedQualityLevel = LAG.OrigQuality
		end)
		LAG.OrigQuality = nil
	end
	if LAG.OrigTech ~= nil then
		pcall(function() Lighting.Technology = LAG.OrigTech end)
		LAG.OrigTech = nil
	end
end

function LAG.Apply()
	LAG.Restore()  -- luôn bắt đầu từ trạng thái gốc (đổi mức không bị chồng)
	local gen = LAG.Gen
	local level = CFG.lagLevel

	-- đồ họa chung: đây là 2 dòng THẬT SỰ giảm tải máy rõ rệt nhất (đường dẫn cũ dùng
	-- settings().Rendering.QualityLevel không phải thuộc tính có thật nên trước đây
	-- không có tác dụng gì — chỉ là pcall nuốt lỗi âm thầm).
	pcall(function()
		local ugs = UserSettings():GetService("UserGameSettings")
		LAG.OrigQuality = ugs.SavedQualityLevel
		ugs.SavedQualityLevel = level >= 2
			and Enum.SavedQualitySetting.QualityLevel1   -- Mạnh: giảm hết cỡ
			or Enum.SavedQualitySetting.QualityLevel4    -- Nhẹ: giảm vừa phải
	end)
	pcall(function()
		LAG.OrigTech = Lighting.Technology
		-- Compatibility = không đổ bóng động, phẳng nhất, nhẹ nhất cho máy yếu
		-- Voxel = vẫn có bóng nhưng nhẹ hơn nhiều so với ShadowMap/Future mặc định
		Lighting.Technology = level >= 2 and Enum.Technology.Compatibility or Enum.Technology.Voxel
	end)
	pcall(lagSet, Lighting, "GlobalShadows", false)
	if level >= 2 then pcall(lagSet, Lighting, "EnvironmentSpecularScale", 0) end

	local terrain = workspace:FindFirstChildOfClass("Terrain")
	if terrain then
		pcall(lagSet, terrain, "WaterWaveSize", 0)
		pcall(lagSet, terrain, "WaterWaveSpeed", 0)
		pcall(lagSet, terrain, "WaterReflectance", 0)
		pcall(lagSet, terrain, "Decoration", false)
	end

	-- vật thể mới xuất hiện sau đó
	local function onAdded(v)
		if LAG.Gen == gen then pcall(lagProcess, v, level) end
	end
	table.insert(LAG.Conns, workspace.DescendantAdded:Connect(onAdded))
	table.insert(LAG.Conns, Lighting.DescendantAdded:Connect(onAdded))

	-- quét nền theo kiểu "duyệt tới đâu xử lý tới đó" (không gọi GetDescendants() dựng
	-- nguyên cả cây workspace thành 1 mảng khổng lồ cùng lúc — đó chính là chỗ gây đứng
	-- hình 1 nhịp lúc vừa bật, nhất là map to nhiều vật thể). Duyệt bằng ngăn xếp thủ công,
	-- xử lý xong một nhóm nhỏ là nghỉ 1 khung hình rồi mới đi tiếp.
	local BATCH_SIZE = 250  -- nhỏ hơn để mỗi lần nghỉ đều đặn hơn, đỡ giật dù tổng thời gian quét lâu hơn chút

	local function walkAndProcess(root)
		local stack = { root }
		local n = 0
		while #stack > 0 do
			if LAG.Gen ~= gen then return false, n end
			local cur = table.remove(stack)
			local ok, children = pcall(function() return cur:GetChildren() end)
			if ok then
				for _, child in ipairs(children) do
					stack[#stack + 1] = child
					pcall(lagProcess, child, level)
					n += 1
					if n % BATCH_SIZE == 0 then task.wait() end
				end
			end
		end
		return true, n
	end

	task.spawn(function()
		local total = 0
		local ok, n = walkAndProcess(Lighting)
		total += n
		if not ok then return end

		local cam = workspace.CurrentCamera
		if cam then
			ok, n = walkAndProcess(cam)
			total += n
			if not ok then return end
		end

		ok, n = walkAndProcess(workspace)
		total += n
		if not ok then return end

		if alive and not booting then
			setStatus("Fix Lag xong • đã xử lý " .. total .. " vật thể")
		end
	end)
end

-- TÍNH NĂNG: FULLBRIGHT (làm sáng toàn map)

-- Chỉ chỉnh thuộc tính ánh sáng/hiệu ứng tối phía máy mình (Lighting, Atmosphere,

local FULLBRIGHT = {

	Gen = 0,

local function fbSet(inst, prop, value)
	local rec = FULLBRIGHT.Orig[inst]
	if not rec then
		rec = {}
		FULLBRIGHT.Orig[inst] = rec
	end
	if rec[prop] == nil then
		local ok, cur = pcall(function() return inst[prop] end)
		if not ok then return end
		rec[prop] = cur
	end
	pcall(function() inst[prop] = value end)
end

-- xử lý các hiệu ứng hay dùng để làm map tối (sương mù ăn theo Atmosphere, lọc màu tối)
local function fbProcessDarkening(inst)
	if inst:IsA("Atmosphere") then
		fbSet(inst, "Density", 0)
		fbSet(inst, "Haze", 0)
		fbSet(inst, "Glare", 0)
	elseif inst:IsA("ColorCorrectionEffect") then
		fbSet(inst, "Brightness", 0.15)
		fbSet(inst, "Contrast", 0)
		fbSet(inst, "Saturation", 0)
		fbSet(inst, "TintColor", Color3.new(1, 1, 1))
	end
end

function FULLBRIGHT.Restore()
	FULLBRIGHT.Gen += 1
	for _, c in ipairs(FULLBRIGHT.Conns) do c:Disconnect() end
	table.clear(FULLBRIGHT.Conns)

	for inst, rec in pairs(FULLBRIGHT.Orig) do
		if inst and (inst.Parent or inst == Lighting) then
			for prop, val in pairs(rec) do
				pcall(function() inst[prop] = val end)
			end
		end
	end
	table.clear(FULLBRIGHT.Orig)
end

function FULLBRIGHT.Apply()
	FULLBRIGHT.Restore()
	local gen = FULLBRIGHT.Gen

	fbSet(Lighting, "Brightness", 3)
	fbSet(Lighting, "ClockTime", 14)
	fbSet(Lighting, "GlobalShadows", false)
	fbSet(Lighting, "OutdoorAmbient", Color3.fromRGB(180, 180, 180))
	fbSet(Lighting, "Ambient", Color3.fromRGB(180, 180, 180))
	fbSet(Lighting, "ExposureCompensation", 1)
	fbSet(Lighting, "EnvironmentDiffuseScale", 1)
	fbSet(Lighting, "EnvironmentSpecularScale", 1)
	fbSet(Lighting, "FogStart", 0)
	fbSet(Lighting, "FogEnd", 100000)

	for _, v in ipairs(Lighting:GetChildren()) do
		pcall(fbProcessDarkening, v)
	end

	-- hiệu ứng tối thêm vào sau (nhiều game đổi theo giờ/sự kiện trong map)
	table.insert(FULLBRIGHT.Conns, Lighting.ChildAdded:Connect(function(v)
		if FULLBRIGHT.Gen == gen then pcall(fbProcessDarkening, v) end
	end))
end

-- TÍNH NĂNG: TỰ ĐỘNG DỊCH CHAT SANG TIẾNG VIỆT
----------------------------------------------------------------
-- Dùng API dịch công khai của Google (không cần key). Chỉ dịch tin nhắn chat người chơi
-- khác gửi, không đụng gì tới gameplay. Chỉ hoạt động với game dùng TextChatService
-- (chat hiện đại của Roblox) — game nào còn dùng hệ chat cũ (bong bóng chữ kiểu cũ) thì
-- tính năng này không can thiệp được vì Roblox không cho script đọc/sửa UI chat kiểu cũ.
local TextChatService = game:GetService("TextChatService")
-- Bảng cache bình thường, KHÔNG dùng weak table ở đây: key là chuỗi text, nếu đặt
-- weak-key thì Lua có thể dọn rác các chuỗi này bất cứ lúc nào (vì không ai khác giữ
-- tham chiếu tới chúng), khiến cache gần như vô tác dụng, dịch lại liên tục tốn request.
local TRANSLATE_CACHE = {}

local function translateToVi(text)
	text = text and text:gsub("^%s+", ""):gsub("%s+$", "") or ""
	if text == "" then return nil end
	if TRANSLATE_CACHE[text] then return TRANSLATE_CACHE[text] end

	local ok, result = pcall(function()
		local url = "https://translate.googleapis.com/translate_a/single?client=gtx&sl=auto&tl=vi&dt=t&q="
			.. HttpService:UrlEncode(text)
		local body = httpGetText(url)
		if not body then return nil end
		local data = HttpService:JSONDecode(body)
		local parts = {}
		for _, seg in ipairs(data[1]) do
			if seg[1] then parts[#parts + 1] = seg[1] end
		end
		local joined = table.concat(parts)
		return joined ~= "" and joined or nil
	end)

	local translated = ok and result or nil
	if translated then TRANSLATE_CACHE[text] = translated end
	return translated
end

-- true khi câu dịch ra gần như y hệt câu gốc (đa phần game đã chat tiếng Việt sẵn) — bỏ qua
-- cho đỡ tốn request, không cần hiện thêm dòng dịch thừa.
local function looksSameIgnoreCase(a, b)
	return a:lower() == b:lower()
end

local translateHook = nil
local function setupAutoTranslate(on)
	if translateHook then
		pcall(function() TextChatService.OnIncomingMessage = nil end)
		translateHook = nil
	end
	if not on then return end

	translateHook = function(message)
		local props = Instance.new("TextChatMessageProperties")
		props.Text = message.Text
		if not CFG.autoTranslate then return props end

		-- QUAN TRỌNG: phải dịch NGAY TRONG callback này (được phép "treo" chờ ở đây,
		-- Roblox tự đợi), KHÔNG được task.spawn ra ngoài rồi return props trước — vì
		-- Roblox lấy giá trị props ngay tại thời điểm hàm return, sửa props sau đó
		-- (từ một luồng khác) sẽ không còn tác dụng gì với tin nhắn đã hiển thị nữa.
		local translated = translateToVi(message.Text)
		if translated and not looksSameIgnoreCase(translated, message.Text) then
			props.Text = message.Text .. "\n🇻🇳 " .. translated
		end
		return props
	end
	TextChatService.OnIncomingMessage = translateHook
end

-- bộ chỉnh số: nhấn 1 lần = 1 bước, giữ nút = chạy lặp
local function addStepper(page, text, key, min, max, step, fmt, apply)
	fmt = fmt or tostring
	CFG[key] = math.clamp(CFG[key], min, max)

	local f = make("Frame", {
		Size = UDim2.new(1, 0, 0, 28),
		BackgroundColor3 = THEME.Panel,
		LayoutOrder = nxt(page),
	}, page)
	corner(f, 8)
	make("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.fromOffset(10, 0),
		Size = UDim2.new(1, -124, 1, 0),
		Text = text,
		Font = Enum.Font.GothamMedium,
		TextSize = 10,
		TextColor3 = THEME.Sub,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, f)

	local function stepBtn(txt, xOff)
		local b = make("TextButton", {
			AnchorPoint = Vector2.new(1, 0.5),
			Position = UDim2.new(1, xOff, 0.5, 0),
			Size = UDim2.fromOffset(26, 18),
			BackgroundColor3 = Color3.fromRGB(46, 50, 66),
			Text = txt,
			Font = Enum.Font.GothamBold,
			TextSize = 12,
			TextColor3 = Color3.new(1, 1, 1),
		}, f)
		corner(b, 5)
		return b
	end

	local plus = stepBtn("+", -6)
	local valBox = make("TextBox", {
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, -34, 0.5, 0),
		Size = UDim2.fromOffset(52, 18),
		BackgroundColor3 = Color3.fromRGB(30, 33, 46),
		Font = Enum.Font.GothamBlack,
		TextSize = 12,
		TextColor3 = THEME.Accent,
		ClearTextOnFocus = false,
		PlaceholderText = tostring(min),
		TextEditable = true,
	}, f)
	corner(valBox, 5)
	local minus = stepBtn("-", -88)

	local function refresh() valBox.Text = fmt(CFG[key]) end

	local function setV(v)
		v = math.clamp(math.floor(v + 0.5), min, max)
		if v == CFG[key] then refresh() return end
		CFG[key] = v
		refresh()
		apply(v)
		save()
	end

	-- gõ số trực tiếp: bấm vào ô, gõ số, nhấn Enter hoặc bấm ra ngoài để áp dụng
	valBox.Focused:Connect(function()
		valBox.Text = tostring(CFG[key])  -- bỏ hậu tố (vd "m") để gõ số thuần cho dễ
	end)
	valBox.FocusLost:Connect(function()
		local n = tonumber((valBox.Text:gsub("[^%-%d%.]", "")))
		if n then
			setV(n)
		else
			refresh()
		end
	end)

	local function holdRepeat(btn, delta)
		local token = 0
		local function isPress(i)
			return i.UserInputType == Enum.UserInputType.MouseButton1
				or i.UserInputType == Enum.UserInputType.Touch
		end
		btn.InputBegan:Connect(function(i)
			if not isPress(i) then return end
			token += 1
			local my = token
			setV(CFG[key] + delta)
			task.delay(0.4, function()
				while alive and token == my do
					setV(CFG[key] + delta)
					task.wait(0.07)
				end
			end)
		end)
		btn.InputEnded:Connect(function(i)
			if isPress(i) then token += 1 end
		end)
	end
	holdRepeat(plus, step)
	holdRepeat(minus, -step)

	controls[key] = { Refresh = refresh, Apply = function() apply(CFG[key]) end }
	refresh()
end

local function refreshItemTpList()

	clearItemTpRows()

			TextColor3 = THEME.Sub,
			LayoutOrder = 1,
		}, itemTpList)
		table.insert(itemTpRows, lbl)
		return
	end

	local list = {}
	for inst, data in pairs(ITEM.Tracked) do
		if data.Part and data.Part.Parent then
			list[#list + 1] = data
		end
	end
	table.sort(list, function(a, b) return (a.Dist or 0) < (b.Dist or 0) end)

			TextColor3 = THEME.Sub,
			LayoutOrder = 1,
		}, itemTpList)
		table.insert(itemTpRows, lbl)
		return
	end

	for i, data in ipairs(list) do
		if i > ITEM_TP_MAX then break end

		local nm = itemName(data.Inst, data.Kind)
		local d = math.floor((data.Dist or 0) + 0.5)

			AutoButtonColor = false,

			TextColor3 = THEME.Text,

		Text = "MỞ",
		Font = Enum.Font.GothamBold,
		TextSize = 11,
		TextColor3 = Color3.fromRGB(5, 20, 12),
	}, f)
	corner(copy, 8)

	local function openLink()
		-- luôn copy link để dự phòng (Roblox hay chặn mở web từ script)
		local copied = type(setclipboard) == "function" and pcall(setclipboard, s.Link)
		local opened = pcall(function()
			GuiService:OpenBrowserWindow(s.Link)
		end)

		if opened then
			setStatus("Đang mở " .. s.Name .. (copied and " (đã copy link dự phòng)" or "..."))
		elseif copied then
			setStatus("Đã copy link " .. s.Name .. ", dán vào trình duyệt")
		else
			setStatus(s.Name .. ": " .. s.Link)
		end

		copy.Text = "OK"
		task.delay(1.5, function()
			if alive then copy.Text = "MỞ" end
		end)
	end

	copy.MouseButton1Click:Connect(openLink)
	f.MouseButton1Click:Connect(openLink)
end

local hubOpen = false

local function showHub(state)
	hubOpen = state
	if state then
		hub.Visible = true
		hubScale.Scale = zoom * 0.85
		TweenService:Create(hubScale, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.Out), { Scale = zoom }):Play()
	else
		local t = TweenService:Create(hubScale, TweenInfo.new(0.15), { Scale = zoom * 0.85 })
		t:Play()
		t.Completed:Once(function()
			if not hubOpen then hub.Visible = false end
		end)
	end
end

toggleHub = function() showHub(not hubOpen) end

local function setMinimized(state)
	minimized = state
	body.Visible = not state
	minBtn.Text = state and "+" or "_"
	tw(hub, 0.25, { Size = UDim2.fromOffset(W, state and H_MIN or H_FULL) })
end

minBtn.MouseButton1Click:Connect(function() setMinimized(not minimized) end)
closeBtn.MouseButton1Click:Connect(function() showHub(false) end)

	if gameProcessed then return end
	if input.KeyCode == TOGGLE_KEY then toggleHub() end
end))

-- gỡ sạch toàn bộ hub: lưu cài đặt, trả mọi thứ về như cũ, xoá GUI
local function unload()
	if not alive then return end
	writeNow()  -- lưu trạng thái hiện tại trước khi tắt các tính năng
	alive = false
	SpeedF.Stop()
	JumpF.Stop()
	NC.Set(false)
	IJ.Set(false)
	flyStop(true)
	LAG.Restore()
	setupAutoTranslate(false)
	FULLBRIGHT.Restore()
	DODGE.Gen += 1
	dodgeClear()
	ESP.Gen += 1
	clearAll()
	ITEM.Gen += 1
	clearItems()
	stopFps()
	for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
	table.clear(conns)
	gui:Destroy()
	_G.VuongHubUnload = nil
end

_G.VuongHubUnload = unload
unloadBtn.MouseButton1Click:Connect(unload)
