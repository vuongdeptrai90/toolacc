-- Safe call - tránh script bị dừng khi có lỗi

local function safeCall(fn, ...)

    local ok, err = pcall(fn, ...)

    if not ok then warn("[SafeCall] " .. tostring(err)) end

    return ok

end

-- Lấy Character & Humanoid & RootPart an toàn

local function getChar()

    local char = LP.Character

    if not char then return nil, nil, nil end

    local hum = char:FindFirstChildOfClass("Humanoid")

    local hrp = char:FindFirstChild("HumanoidRootPart")

    return char, hum, hrp

end

-- Kiểm tra nhân vật còn sống không

local function isAlive()

    local _, hum = getChar()

    return hum and hum.Health > 0

end

-- Teleport mượt bằng TweenService

local function tweenTo(position, speed)

    local _, _, hrp = getChar()

    if not hrp then return end

    speed = speed or 200

    local distance = (hrp.Position - position).Magnitude

    local time = distance / speed

    local tween = TweenService:Create(

        hrp,

        TweenInfo.new(time, Enum.EasingStyle.Linear),

        { CFrame = CFrame.new(position) }

    )

    tween:Play()

    return tween

end

-- Teleport tức thời (dùng cho Snipe Fruit / Island)

local function teleportInstant(position)

    local _, _, hrp = getChar()

    if not hrp then return end

    hrp.CFrame = CFrame.new(position + Vector3.new(0, 5, 0))

end

-- Trang bị vũ khí theo loại

local function equipWeapon(wType)

    local char = LP.Character

    if not char then return end

    local backpack = LP:FindFirstChild("Backpack")

    if not backpack then return end

    local keyWords = {

        Melee     = { "Melee", "Fighting", "Combat" },

        Sword     = { "Sword", "Saber", "Blade", "Katana" },

        Gun       = { "Gun", "Pistol", "Rifle", "Bow" },

        BloxFruit = { "Fruit", "FruitBlox" },

    }

    local keys = keyWords[wType] or { wType }

    for _, tool in pairs(backpack:GetChildren()) do

        if tool:IsA("Tool") then

            for _, keyword in ipairs(keys) do

                if tool.Name:lower():find(keyword:lower()) then

                    safeCall(function()

                        char.Humanoid:EquipTool(tool)

                    end)

                    return

                end

            end

        end

    end

end

-- Lấy trái ác quỷ đang cầm

local function getCurrentFruit()

    local char = LP.Character

    if not char then return nil end

    for _, tool in pairs(char:GetChildren()) do

        if tool:IsA("Tool") and tool:FindFirstChild("Fruit") then

            return tool

        end

    end

    return nil

end

-- Đánh quái (sử dụng tool hiện tại + remote combat)

local function attack()

    local tool = LP.Character and LP.Character:FindFirstChildOfClass("Tool")

    if not tool then return end

    -- Tool-based attack

    if tool:FindFirstChild("Click") or tool:FindFirstChild("Activated") then

        safeCall(function() tool:Activate() end)

    end

    -- Remote attack fallback

    local combatRemote = ReplicatedStorage:FindFirstChild("Combat")

    if combatRemote then

        safeCall(function()

            combatRemote:FireServer("Attack")

        end)

    end

end

-- Tìm quái gần nhất trong phạm vi

local function findNearestEnemy(maxDist)

    local _, _, hrp = getChar()

    if not hrp then return nil end

    local nearest, shortest = nil, maxDist or Config.FarmDistance

    for _, obj in pairs(Workspace:GetDescendants()) do

        if obj:IsA("Model") and obj:FindFirstChild("Humanoid") and obj ~= LP.Character then

            local hum = obj:FindFirstChild("Humanoid")

            local root = obj:FindFirstChild("HumanoidRootPart") or obj.PrimaryPart

            if hum and root and hum.Health > 0 then

                -- Bỏ qua người chơi khác

                local isPlayer = Players:GetPlayerFromCharacter(obj)

                if not isPlayer then

                    local dist = (hrp.Position - root.Position).Magnitude

                    if dist < shortest then

                        shortest = dist

                        nearest = obj

                    end

                end

            end

        end

    end

    return nearest, shortest

end

local espCache = {}

local function createFruitESP(fruit)

    if espCache[fruit] then return end

    local billboard = Instance.new("BillboardGui")

    billboard.Name = "[A]_FruitESP"

    billboard.Adornee = fruit

    billboard.AlwaysOnTop = true

    billboard.Size = UDim2.new(0, 100, 0, 40)

    billboard.StudsOffset = Vector3.new(0, 3, 0)

    local label = Instance.new("TextLabel")

    label.Parent = billboard

    label.BackgroundTransparency = 1

    label.Size = UDim2.new(1, 0, 1, 0)

    label.Text = "🍎 " .. fruit.Name

    label.TextColor3 = Color3.fromRGB(255, 100, 100)

    label.TextStrokeTransparency = 0

    label.TextScaled = true

    label.Font = Enum.Font.GothamBold

    billboard.Parent = fruit

    espCache[fruit] = billboard

end

local function removeFruitESP(fruit)

    if espCache[fruit] then

        espCache[fruit]:Destroy()

        espCache[fruit] = nil

    end

end

-- Xử lý Auto Farm chung

local function processFarm()

    if not isAlive() then return end

    -- Safe Mode check

    if Flags.SafeMode then

        local _, hum = getChar()

        if hum and hum.Health / hum.MaxHealth * 100 < Config.SafeHP then

            return

        end

    end

    -- Tự động trang bị vũ khí

    equipWeapon(Config.WeaponType)

    -- Tìm quái gần nhất

    local enemy, dist = findNearestEnemy(Config.FarmDistance)

    if enemy then

        local root = enemy:FindFirstChild("HumanoidRootPart") or enemy.PrimaryPart

        if root then

            -- Di chuyển đến quái (dùng tween nếu xa)

            if dist > 15 then

                tweenTo(root.Position, 150)

            end

            task.wait(0.1)

            -- Đánh quái

            attack()

            task.wait(Flags.FastAttack and 0.05 or Config.AttackSpeed)

        end

    end

end

-- Xử lý Auto Stats

local function processStats()

    if not isAlive() then return end

    local stats = LP:FindFirstChild("leaderstats")

    if not stats then return end

    local points = stats:FindFirstChild("Points") or stats:FindFirstChild("StatPoints")

    if not points or points.Value <= 0 then return end

    -- Remote add stat

    local addStat = ReplicatedStorage:FindFirstChild("AddStat") or ReplicatedStorage:FindFirstChild("CommF_")

    if addStat then

        local order = { "Melee", "Defense", "Sword", "Gun", "BloxFruit" }

        for _, statName in ipairs(order) do

            if Config.StatPoints[statName] > 0 then

                safeCall(function()

                    addStat:InvokeServer(statName)

                end)

                task.wait(0.3)

                return

            end

        end

    end

end

-- Xử lý Auto Store Fruit

local function processStoreFruit()

    local fruit = getCurrentFruit()

    if not fruit then return end

    -- Gửi remote store

    local storeRemote = ReplicatedStorage:FindFirstChild("StoreFruit")

    if storeRemote then

        safeCall(function() storeRemote:InvokeServer(fruit.Name) end)

    end

end

-- Xử lý Auto Bring Fruit

local function processBringFruit()

    local _, _, hrp = getChar()

    if not hrp then return end

    for _, obj in pairs(Workspace:GetDescendants()) do

        if obj:IsA("Tool") and obj.Name:lower():find("fruit") and obj:FindFirstChild("Handle") then

            teleportInstant(obj.Handle.Position)

            return

        end

    end

end

-- Vòng lặp chính

task.spawn(function()

    while true do

        task.wait(0.1)

        safeCall(function()

            -- Farm chung

            if Flags.AutoFarmLevel or Flags.AutoFarmBone or Flags.AutoFarmBoss or Flags.AutoFarmAllBoss then

                processFarm()

            end

            -- Auto Stats

            if Flags.AutoStats then

                processStats()

            end

            -- Auto Store Fruit

            if Flags.AutoStoreFruit then

                processStoreFruit()

            end

           if Flags.AutoBringFruit then
           processBringFruit()
end
end)
end
end)
